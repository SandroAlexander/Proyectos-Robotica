const int ledPin = 13; // LED EN PIN 13
bool flagEncendido = 0;

void setup() {
  pinMode(ledPin, OUTPUT);         // LED como salida
}

void loop() {
  // Encender el LED por un segundo si se pulsa el boton

  if (flagEncendido) {
    digitalWrite(ledPin, LOW);
    flagEncendido = 0;
  }
  else {
  	digitalWrite(ledPin, HIGH);
    flagEncendido = 1;
  }
  delay(200);
}
